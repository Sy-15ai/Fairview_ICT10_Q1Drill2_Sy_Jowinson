Arithmetic Operations

Enter two numbers to calculate their results.

from js import document from pyscript import display def calculate(event): # Get the first number from the first text box. num1 = float(document.getElementById("num1").value) # Get the second number from the second text box. num2 = float(document.getElementById("num2").value) # Addition: Adds the first number and second number. addition = num1 + num2 # Subtraction: Subtracts the second number from the first number. subtraction = num1 - num2 # Multiplication: Multiplies the first number by the second number. multiplication = num1 * num2 # Division: Divides the first number by the second number. if num2 != 0: division = num1 / num2 else: division = "Cannot divide by zero" # Modulus: Finds the remainder after division. if num2 != 0: modulus = num1 % num2 else: modulus = "Cannot calculate modulus by zero" # Exponentiation: Raises the first number to the power of the second number. exponentiation = num1 ** num2 # Floor Division: Divides the numbers and rounds the result down. if num2 != 0: floor_division = num1 // num2 else: floor_division = "Cannot divide by zero" # Create the results using an HTML f-string. results = f"""

Results

Addition: {addition}

Subtraction: {subtraction}

Multiplication: {multiplication}

Division: {division}

Modulus: {modulus}

Exponentiation: {exponentiation}

Floor Division: {floor_division}

""" # Display the HTML results in the output area. output = document.getElementById("output") output.innerHTML = results # Use display() with the f-string as required. display(results)
